```typescript
export abstract class BaseParser {
  abstract parse(content: string): ParseResult;
```

```typescript
import { BaseParser } from "./base.js";
import type { NormalizedMessage, ParseResult, MessageRole } from "../types/index.js";

/**
 * Claude Code JSONL Parser
 * 
 * 输入格式示例：
 * {"type":"user","content":"帮我创建一个项目"}
 * {"type":"assistant","content":"好的，我来帮你..."}
 * {"type":"tool_use","tool_use_id":"abc123","tool_name":"write_file","content":"..."}
 * {"type":"tool_result","tool_use_id":"abc123","content":"File written"}
 */
export class ClaudeCodeParser extends BaseParser {
  parse(content: string): ParseResult {
    const lines = content.trim().split("\n").filter(line => line.trim());
    const messages: NormalizedMessage[] = [];

    for (let i = 0; i < lines.length; i++) {
      const line = lines[i];
      try {
        const raw = JSON.parse(line);
        const msg = this.normalizeMessage(raw, i);
        if (msg) messages.push(msg);
      } catch (error) {
        console.error(`Failed to parse line ${i}: ${error}`);
      }
    }

    return { messages };
  }

  private normalizeMessage(raw: any, index: number): NormalizedMessage | null {
    const typeMapping: Record<string, MessageRole> = {
      "system": "system",
      "user": "user",
      "assistant": "assistant",
      "tool_use": "tool",
      "tool_result": "tool",
    };

    const role = typeMapping[raw.type];
    if (!role) return null;

    const msg: NormalizedMessage = {
      id: this.generateId(),
      index,
      role,
      content: raw.content || "",
      tokens: this.estimateTokens(raw.content || ""),
      timestamp: raw.timestamp,
    };

    // 处理工具调用
    if (raw.type === "tool_use") {
      msg.isToolUse = true;
      msg.toolUseId = raw.tool_use_id;
      msg.toolName = raw.tool_name;
    }

    if (raw.type === "tool_result") {
      msg.isToolResult = true;
      msg.toolUseId = raw.tool_use_id;
    }

    return msg;
  }
}
```

### 3.3 文件：`src/parsers/openai.ts`

```typescript
import { BaseParser } from "./base.js";
import type { NormalizedMessage, ParseResult, MessageRole } from "../types/index.js";

/**
 * OpenAI JSONL Parser
 * 
 * 输入格式示例：
 * {"messages":[{"role":"system","content":"You are..."},{"role":"user","content":"..."}]}
 */
export class OpenAIParser extends BaseParser {
  parse(content: string): ParseResult {
    const lines = content.trim().split("\n").filter(line => line.trim());
    const messages: NormalizedMessage[] = [];

    let globalIndex = 0;

    for (const line of lines) {
      try {
        const raw = JSON.parse(line);
        
        // OpenAI 格式可能是单条消息或 messages 数组
        const msgList = raw.messages ? raw.messages : [raw];
        
        for (const rawMsg of msgList) {
          const msg = this.normalizeMessage(rawMsg, globalIndex++);
          if (msg) messages.push(msg);
        }
      } catch (error) {
        console.error(`Failed to parse: ${error}`);
      }
    }

    return { messages };
  }

  private normalizeMessage(raw: any, index: number): NormalizedMessage | null {
    if (!raw.role || !raw.content) return null;

    const msg: NormalizedMessage = {
      id: this.generateId(),
      index,
      role: raw.role as MessageRole,
      content: raw.content,
      tokens: this.estimateTokens(raw.content),
      timestamp: raw.timestamp,
    };

    // 处理 function call
    if (raw.function_call) {
      msg.isToolUse = true;
      msg.toolName = raw.function_call.name;
      msg.toolUseId = raw.function_call.id || this.generateId();
      msg.content = JSON.stringify(raw.function_call);
    }

    // 处理 tool calls (新格式)
    if (raw.tool_calls) {
      msg.isToolUse = true;
      msg.toolName = raw.tool_calls[0]?.function?.name;
      msg.toolUseId = raw.tool_calls[0]?.id;
    }

    return msg;
  }
}
```

### 4.1 文件：`src/utils/tokenizer.ts`

```typescript
/**
 * Approximate Tokenizer
 * 不调用外部 API，基于规则的 token 估算
 * 
 * 规则：
 * 1. 英文：~4 字符 = 1 token
 * 2. 中文：~1.5 字符 = 1 token
 * 3. 代码：~3 字符 = 1 token
 * 4. 空白字符：忽略
 */

export interface TokenizerResult {
  tokens: number;
  breakdown: {
    english: number;
    chinese: number;
    code: number;
    whitespace: number;
  };
}

/**
 * 近似 token 计数
 * @param content 输入文本
 * @returns token 数量
 */
export function approximateTokens(content: string): number {
  if (!content) return 0;
  
  const result = analyzeContent(content);
  
  // 加权计算
  const tokens = 
    Math.ceil(result.english / 4) +
    Math.ceil(result.chinese / 1.5) +
    Math.ceil(result.code / 3);
  
  return Math.max(1, tokens);
}

/**
 * 详细分析
 */
export function analyzeContent(content: string): TokenizerResult["breakdown"] {
  let english = 0;
  let chinese = 0;
  let code = 0;
  let whitespace = 0;
  
  // 检测是否在代码块内
  const inCodeBlock = /```[\s\S]*?```/g;
  const codeBlocks = content.match(inCodeBlock) || [];
  const codeContent = codeBlocks.join("");
  
  // 移除代码块后的内容
  const nonCodeContent = content.replace(inCodeBlock, "");
  
  // 分析代码块
  for (const char of codeContent) {
    if (/\s/.test(char)) {
      whitespace++;
    } else {
      code++;
    }
  }
  
  // 分析非代码内容
  for (const char of nonCodeContent) {
    if (/\s/.test(char)) {
      whitespace++;
    } else if (/[\u4e00-\u9fa5]/.test(char)) {
      chinese++;
    } else if (/[a-zA-Z0-9]/.test(char)) {
      english++;
    } else {
      // 符号按英文处理
      english++;
    }
  }
  
  return { english, chinese, code, whitespace };
}

/**
 * 精确模式（用于测试）
 * 使用更细致的规则
 */
export function preciseTokenCount(content: string): number {
  const lines = content.split("\n");
  let total = 0;
  
  for (const line of lines) {
    // 空行
    if (line.trim().length === 0) {
      total += 1;
      continue;
    }
    
    // 代码行（检测缩进）
    if (/^(\s{2,}|\t)/.test(line)) {
      total += Math.ceil(line.length / 3);
      continue;
    }
    
    // 普通文本
    total += approximateTokens(line);
  }
  
  return total;
}
```

```typescript
/**

import type { NormalizedMessage } from "../types/index.js";

export interface ProtectionResult {
  isProtected: boolean;
  reason?: string;
}

// 最近 N 条消息强制保护
const RECENT_MESSAGES_WINDOW = 6;  // trimctx 要求：6 轮

// 保护规则列表
export function checkProtection(
  msg: NormalizedMessage,
  allMessages: NormalizedMessage[],
  totalCount: number
): ProtectionResult {

  // ━━━ 规则 1：system/developer 消息 ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  if (msg.role === "system" || msg.role === "developer") {
    return { isProtected: true, reason: "system_or_developer_message" };
  }

  // ━━━ 规则 2：最近 6 轮消息 ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  const fromEnd = totalCount - 1 - msg.index;
  if (fromEnd < RECENT_MESSAGES_WINDOW) {
    return { isProtected: true, reason: `recent_${fromEnd + 1}_messages` };
  }

  // ━━━ 规则 3：包含代码块 ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  if (containsCodeBlock(msg.content)) {
    return { isProtected: true, reason: "contains_code_block" };
  }

  // ━━━ 规则 4：包含错误栈 ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  if (containsErrorStack(msg.content)) {
    return { isProtected: true, reason: "contains_error_stack" };
  }

  // ━━━ 规则 5：包含文件路径 ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  if (containsFilePath(msg.content)) {
    return { isProtected: true, reason: "contains_file_path" };
  }

  // ━━━ 规则 6：包含 shell 命令 ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  if (containsShellCommand(msg.content)) {
    return { isProtected: true, reason: "contains_shell_command" };
  }

  // ━━━ 规则 7：记忆类指令 ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  // ... 保留原有逻辑 ...

  return { isProtected: false };
}

}

function containsErrorStack(content: string): boolean {
  const patterns = [
    /Error:\s+.+/,
    /Exception:\s+.+/,
    /at\s+\w+\.\w+\(.+:\d+:\d+\)/,  // 堆栈跟踪
    /Stack trace:/i,
    /Traceback \(most recent call last\)/,
    /\#\d+\s+0x[0-9a-f]+/,  // C++ 堆栈
  ];
  
  return patterns.some(p => p.test(content));
}

function containsFilePath(content: string): boolean {
  // 匹配常见文件路径模式
  const patterns = [
    /(?:^|[\s'"`])([./~][\w./\-@]+\.(?:ts|tsx|js|jsx|py|json|yaml|yml|md|css|html|sh|env|txt))/gm,
```

### 6.1 文件：`src/compressor/index.ts`

```typescript
import type { NormalizedMessage, SessionReport } from "../types/index.js";

export interface CompressOptions {
  removeRotThreshold?: number;    // 默认 0.75
  compressRotThreshold?: number;  // 默认 0.50
  preserveProtected?: boolean;    // 默认 true
}

/**
 * 压缩会话
 * 根据 analysis report 决定保留/压缩/删除哪些消息
 */
export function compressSession(
  messages: NormalizedMessage[],
  report: SessionReport,
  options: CompressOptions = {}
): NormalizedMessage[] {
  const {
    preserveProtected = true,
  } = options;

  const result: NormalizedMessage[] = [];
  
  // 构建决策映射
  const decisionMap = new Map<string, typeof report.analyses[0]["decision"]>();
  for (const analysis of report.analyses) {
    decisionMap.set(analysis.messageId, analysis.decision);
  }

  for (const msg of messages) {
    const decision = decisionMap.get(msg.id);
    
    if (!decision) {
      // 未分析的消息，保守保留
      result.push(msg);
      continue;
    }

    switch (decision) {
      case "keep_protected":
        if (preserveProtected) {
          result.push(msg);
        }
        break;
        
      case "keep":
        result.push(msg);
        break;
        
      case "compress_candidate":
        // 压缩策略：保留摘要
        const compressed = compressMessage(msg);
        result.push(compressed);
        break;
        
      case "remove_candidate":
        // 直接跳过（不添加到结果）
        break;
    }
  }

  // 重新索引
  return result.map((msg, idx) => ({ ...msg, index: idx }));
}

/**
 * 压缩单条消息
 * 保留关键信息，去除冗余
 */
function compressMessage(msg: NormalizedMessage): NormalizedMessage {
  const lines = msg.content.split("\n");
  
  // 策略 1：保留前 3 行和后 3 行
  if (lines.length > 10) {
    const compressed = [
      ...lines.slice(0, 3),
      `... [${lines.length - 6} lines compressed] ...`,
      ...lines.slice(-3),
    ].join("\n");
    
    return {
      ...msg,
      content: compressed,
      tokens: Math.floor(msg.tokens * 0.4),  // 估算压缩后 token
    };
  }
  
  // 策略 2：短消息直接保留
  return msg;
}

/**
 * 估算压缩率
 */
export function estimateCompressionRatio(report: SessionReport): number {
  const { summary } = report;
  
  const totalTokens = summary.total_tokens;
  const removeTokens = report.analyses
    .filter(a => a.decision === "remove_candidate")
    .reduce((sum, a) => sum + a.tokens, 0);
  
  const compressTokens = report.analyses
    .filter(a => a.decision === "compress_candidate")
    .reduce((sum, a) => sum + a.tokens * 0.6, 0);  // 压缩节省 60%
  
  const savings = removeTokens + compressTokens;
  
  return totalTokens > 0 ? savings / totalTokens : 0;
}
```

```typescript
import { readFileSync, writeFileSync } from "fs";
import { ClaudeCodeParser, OpenAIParser, detectParserType } from "../../parsers/index.js";
import { analyzeSession } from "../../analyzers/decisionEngine.js";
import { compressSession } from "../../compressor/index.js";

export function compressCommand(
  file: string, 
  options: { output?: string; parser: string }
) {
  if (!options.output) {
    console.error("❌ Output file path required. Use -o option.");
    process.exit(1);
  }
  
  console.log(`\n🗜️  Compressing: ${file}\n`);
  
  const content = readFileSync(file, "utf-8");
  
  const parserType = options.parser === "auto" 
    ? detectParserType(content) 
    : options.parser;
  
  if (!parserType) {
    console.error("❌ Could not detect parser type.");
    process.exit(1);
  }
  
  const parser = parserType === "claude-code" 
    ? new ClaudeCodeParser() 
    : new OpenAIParser();
  
  const { messages } = parser.parse(content);
  
  console.log(`  Original messages: ${messages.length}`);
  
  // 分析
  const report = analyzeSession(messages, file);
  
  // 压缩
  const compressed = compressSession(messages, report);
  
  console.log(`  Compressed messages: ${compressed.length}`);
  
  // 写入新文件（不修改原文件）
  const outputContent = compressed
    .map(msg => JSON.stringify(msg))
    .join("\n");
  
  writeFileSync(options.output, outputContent, "utf-8");
  
  console.log(`\n✅ Compressed file saved to: ${options.output}`);
  console.log(`⚠️  Original file unchanged: ${file}\n`);
}
```
